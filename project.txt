package com.televault.app

import android.util.Base64
import java.io.DataInputStream
import java.io.DataOutputStream
import java.io.File
import java.io.FileInputStream
import java.io.FileOutputStream
import java.io.InputStream
import java.security.SecureRandom
import javax.crypto.Cipher
import javax.crypto.KeyGenerator
import javax.crypto.spec.GCMParameterSpec
import javax.crypto.spec.SecretKeySpec

object CryptoHelper {
    private const val CHUNK_SIZE = 1024 * 1024
    private const val NONCE_SIZE = 12
    private const val TAG_BITS = 128

    fun generateAes256KeyBase64(): String {
        val kg = KeyGenerator.getInstance("AES")
        kg.init(256)
        return Base64.encodeToString(kg.generateKey().encoded, Base64.NO_WRAP)
    }

    fun generateApiKey(): String {
        val bytes = ByteArray(24)
        SecureRandom().nextBytes(bytes)
        val token = Base64.encodeToString(bytes, Base64.URL_SAFE or Base64.NO_WRAP or Base64.NO_PADDING)
        return "tv_live_$token"
    }

    fun encryptStreamToFile(input: InputStream, outFile: File, keyBase64: String) {
        val keySpec = SecretKeySpec(Base64.decode(keyBase64, Base64.NO_WRAP), "AES")
        val random = SecureRandom()
        val buffer = ByteArray(CHUNK_SIZE)

        DataOutputStream(FileOutputStream(outFile).buffered()).use { dos ->
            while (true) {
                val read = input.read(buffer)
                if (read <= 0) break
                val nonce = ByteArray(NONCE_SIZE)
                random.nextBytes(nonce)

                val cipher = Cipher.getInstance("AES/GCM/NoPadding")
                cipher.init(Cipher.ENCRYPT_MODE, keySpec, GCMParameterSpec(TAG_BITS, nonce))
                val encrypted = cipher.doFinal(buffer, 0, read)

                dos.writeInt(encrypted.size)
                dos.write(nonce)
                dos.write(encrypted)
            }
        }
    }

    fun decryptFile(encFile: File, outFile: File, keyBase64: String) {
        val keySpec = SecretKeySpec(Base64.decode(keyBase64, Base64.NO_WRAP), "AES")

        DataInputStream(FileInputStream(encFile).buffered()).use { dis ->
            FileOutputStream(outFile).buffered().use { fos ->
                while (dis.available() > 0) {
                    val encLen = dis.readInt()
                    val nonce = ByteArray(NONCE_SIZE)
                    dis.readFully(nonce)
                    val encrypted = ByteArray(encLen)
                    dis.readFully(encrypted)

                    val cipher = Cipher.getInstance("AES/GCM/NoPadding")
                    cipher.init(Cipher.DECRYPT_MODE, keySpec, GCMParameterSpec(TAG_BITS, nonce))
                    val plain = cipher.doFinal(encrypted)
                    fos.write(plain)
                }
            }
        }
    }
}

package com.televault.app

import android.content.ContentValues
import android.content.Context
import android.database.sqlite.SQLiteDatabase
import android.database.sqlite.SQLiteOpenHelper
import java.io.File

data class ProjectDb(
    val id: Long,
    val name: String,
    val tgChatId: Long,
    val apiKey: String,
    val encKey: String,
    val createdAt: Long
)

data class VaultFile(
    val id: Long,
    val projectId: Long,
    val fileName: String,
    val tgMessageId: Long,
    val tgFileId: Int,
    val fileSize: Long,
    val createdAt: Long

class VaultDbHelper(private val context: Context) :
    SQLiteOpenHelper(context, DB_NAME, null, DB_VERSION) {

    companion object {
        const val DB_NAME = "televault_master.db"
        const val DB_VERSION = 1
    }

    override fun onCreate(db: SQLiteDatabase) {
        db.execSQL("""
            CREATE TABLE projects (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                name TEXT NOT NULL,
                tg_chat_id INTEGER NOT NULL,
                api_key TEXT NOT NULL,
                enc_key TEXT NOT NULL,
                created_at INTEGER NOT NULL
            )
        """)
        db.execSQL("""
            CREATE TABLE vault_files (
                id INTEGER PRIMARY KEY AUTOINCREMENT,
                project_id INTEGER NOT NULL,
                file_name TEXT NOT NULL,
                tg_message_id INTEGER NOT NULL,
                tg_file_id INTEGER NOT NULL,
                file_size INTEGER NOT NULL,
                created_at INTEGER NOT NULL
            )
        """)
    }

    override fun onUpgrade(db: SQLiteDatabase, oldVersion: Int, newVersion: Int) {}

    fun insertProject(name: String, tgChatId: Long, apiKey: String, encKey: String): Long {
        val cv = ContentValues().apply {
            put("name", name)
            put("tg_chat_id", tgChatId)
            put("api_key", apiKey)
            put("enc_key", encKey)
            put("created_at", System.currentTimeMillis())
        }
        return writableDatabase.insert("projects", null, cv)
    }

    fun getAllProjects(): List<ProjectDb> {
        val list = mutableListOf<ProjectDb>()
        readableDatabase.rawQuery(
            "SELECT id, name, tg_chat_id, api_key, enc_key, created_at FROM projects ORDER BY id DESC",
            null
        ).use { c ->
            while (c.moveToNext()) {
                list.add(
                    ProjectDb(
                        id = c.getLong(0),
                        name = c.getString(1),
                        tgChatId = c.getLong(2),
                        apiKey = c.getString(3),
                        encKey = c.getString(4),
                        createdAt = c.getLong(5)
                    )
                )
            }
        }
        return list
    }

    fun insertFile(projectId: Long, fileName: String, tgMsgId: Long, tgFileId: Int, size: Long): Long {
        val cv = ContentValues().apply {
            put("project_id", projectId)
            put("file_name", fileName)
            put("tg_message_id", tgMsgId)
            put("tg_file_id", tgFileId)
            put("file_size", size)
            put("created_at", System.currentTimeMillis())
        }
        return writableDatabase.insert("vault_files", null, cv)
    }

    fun searchFiles(projectId: Long, query: String): List<VaultFile> {
        val list = mutableListOf<VaultFile>()
        readableDatabase.rawQuery(
            "SELECT id, project_id, file_name, tg_message_id, tg_file_id, file_size, created_at FROM vault_files WHERE project_id = ? AND file_name LIKE ? ORDER BY id DESC",
            arrayOf(projectId.toString(), "%$query%")
        ).use { c ->
            while (c.moveToNext()) {
                list.add(
                    VaultFile(
                        id = c.getLong(0),
                        projectId = c.getLong(1),
                        fileName = c.getString(2),
                        tgMessageId = c.getLong(3),
                        tgFileId = c.getInt(4),
                        fileSize = c.getLong(5),
                        createdAt = c.getLong(6)
                    )
                )
            }
        }
        return list
    }

    fun getDbFile(): File = context.getDatabasePath(DB_NAME)
}

package com.televault.app

import android.content.Context
import android.os.Build
import org.drinkless.td.libcore.telegram.Client
import org.drinkless.td.libcore.telegram.TdApi
import java.io.File
import java.util.concurrent.ConcurrentHashMap

class TelegramManager(
    private val context: Context,
    private val onAuthState: (String) -> Unit
) {
    private var client: Client? = null
    private var apiId: Int = 0
    private var apiHash: String = ""

    private val pendingUploads = ConcurrentHashMap<Long, (Long, Int) -> Unit>()
    private val pendingDownloads = ConcurrentHashMap<Int, (String) -> Unit>()

    fun start(apiId: Int, apiHash: String) {
        this.apiId = apiId
        this.apiHash = apiHash
        if (client != null) return

        client = Client.create(
            { obj -> handleUpdate(obj) },
            null,
            null
        )
        client?.send(TdApi.SetLogVerbosityLevel(1), null)
    }

    private fun handleUpdate(obj: TdApi.Object) {
        when (obj) {
            is TdApi.UpdateAuthorizationState -> {
                when (obj.authorizationState) {
                    is TdApi.AuthorizationStateWaitTdlibParameters -> sendParameters()
                    is TdApi.AuthorizationStateWaitPhoneNumber -> onAuthState("WAIT_PHONE")
                    is TdApi.AuthorizationStateWaitCode -> onAuthState("WAIT_CODE")
                    is TdApi.AuthorizationStateWaitPassword -> onAuthState("WAIT_PASSWORD")
                    is TdApi.AuthorizationStateReady -> onAuthState("READY")
                    is TdApi.AuthorizationStateClosed -> onAuthState("CLOSED")
                }
            }

            is TdApi.UpdateMessageSendSucceeded -> {
                val callback = pendingUploads.remove(obj.oldMessageId)
                if (callback != null) {
                    val msg = obj.message
                    val doc = msg.content as? TdApi.MessageDocument
                    val fileId = doc?.document?.document?.id ?: 0
                    callback(msg.id, fileId)
                }
            }

            is TdApi.UpdateFile -> {
                val f = obj.file
                if (f.local.isDownloadingCompleted && f.local.path.isNotEmpty()) {
                    pendingDownloads.remove(f.id)?.invoke(f.local.path)
                }
            }
        }
    }

    private fun sendParameters() {
        val tdDir = File(context.filesDir, "tdlib")
        val params = TdApi.SetTdlibParameters().apply {
            useTestDc = false
            databaseDirectory = File(tdDir, "db").absolutePath
            filesDirectory = File(tdDir, "files").absolutePath
            databaseEncryptionKey = ByteArray(0)
            useFileDatabase = true
            useChatInfoDatabase = true
            useMessageDatabase = true
            useSecretChats = false
            apiId = this@TelegramManager.apiId
            apiHash = this@TelegramManager.apiHash
            systemLanguageCode = "en"
            deviceModel = Build.MODEL
            systemVersion = Build.VERSION.RELEASE
            applicationVersion = "1.0"
        }
        client?.send(params, null)
    }

    fun sendPhone(phone: String, onError: (String) -> Unit) {
        client?.send(TdApi.SetAuthenticationPhoneNumber(phone, null)) { res ->
            if (res is TdApi.Error) onError(res.message)
        }
    }

    fun sendOtp(code: String, onError: (String) -> Unit) {
        client?.send(TdApi.CheckAuthenticationCode(code)) { res ->
            if (res is TdApi.Error) onError(res.message)
        }
    }

    fun send2FaPassword(password: String, onError: (String) -> Unit) {
        client?.send(TdApi.CheckAuthenticationPassword(password)) { res ->
            if (res is TdApi.Error) onError(res.message)
        }
    }

    fun createPrivateChannel(title: String, onResult: (Long?, String?) -> Unit) {
        val req = TdApi.CreateNewSupergroupChat().apply {
            this.title = "TV_DB_$title"
            this.isForum = false
            this.isChannel = true
            this.description = "TeleVault Private Cloud Database"
        }
        client?.send(req) { res ->
            when (res) {
                is TdApi.Chat -> onResult(res.id, null)
                is TdApi.Error -> onResult(null, res.message)
                else -> onResult(null, "অজানা ত্রুটি")
            }
        }
    }

    fun uploadFile(chatId: Long, file: File, caption: String, onDone: (Long?, Int?, String?) -> Unit) {
        val msgContent = TdApi.InputMessageDocument().apply {
            document = TdApi.InputFileLocal(file.absolutePath)
            this.caption = TdApi.FormattedText(caption, emptyArray())
        }
        val req = TdApi.SendMessage().apply {
            this.chatId = chatId
            this.inputMessageContent = msgContent
        }
        client?.send(req) { res ->
            when (res) {
                is TdApi.Message -> {
                    pendingUploads[res.id] = { finalMsgId, finalFileId ->
                        onDone(finalMsgId, finalFileId, null)
                    }
                }
                is TdApi.Error -> onDone(null, null, res.message)
            }
        }
    }

    fun downloadFile(fileId: Int, onDone: (File?, String?) -> Unit) {
        pendingDownloads[fileId] = { path -> onDone(File(path), null) }
        val req = TdApi.DownloadFile(fileId, 32, 0, 0, true)
        client?.send(req) { res ->
            if (res is TdApi.Error) {
                pendingDownloads.remove(fileId)
                onDone(null, res.message)
            }
        }
    }
}

    private fun backupSqliteToSelectedChannel() {
        val proj = selectedProject ?: return toast("একটি ডেটাবেস সিলেক্ট করুন")
        Thread {
            try {
                val dbFile = db.getDbFile()
                val encBackup = File(cacheDir, "backup_${System.currentTimeMillis()}.db.vault")
                dbFile.inputStream().use { input ->
                    CryptoHelper.encryptStreamToFile(input, encBackup, proj.encKey)
                }
                tg.uploadFile(proj.tgChatId, encBackup, "SQLite Backup") { msgId, _, err ->
                    runOnUiThread {
                        if (msgId != null) toast("SQLite ব্যাকআপ টেলিগ্রামে সেভ হয়েছে!")
                        else toast("ব্যাকআপ ব্যর্থ: $err")
                    }
                }
            } catch (e: Exception) {
                runOnUiThread { toast("ত্রুটি: ${e.message}") }
            }
        }.start()
    }

    private fun refreshFiles() {
        filesContainer.removeAllViews()
        val proj = selectedProject ?: return
        val q = searchInput.text.toString().trim()
        val files = db.searchFiles(proj.id, q)
        files.forEach { f ->
            filesContainer.addView(Button(this).apply {
                text = "⬇️ ${f.fileName} (${f.fileSize / 1024} KB) — আনলক করুন"
                setOnClickListener { downloadAndDecrypt(f, proj) }
            })
        }
    }

    private fun downloadAndDecrypt(vf: VaultFile, proj: ProjectDb) {
        toast("ডাউনলোড ও আনলক হচ্ছে...")
        tg.downloadFile(vf.tgFileId) { encFile, err ->
            if (encFile == null) {
                runOnUiThread { toast("ডাউনলোড ব্যর্থ: $err") }
                return@downloadFile
            }
            Thread {
                try {
                    val outDir = File(filesDir, "unlocked").apply { mkdirs() }
                    val outFile = File(outDir, vf.fileName)
                    CryptoHelper.decryptFile(encFile, outFile, proj.encKey)
                    runOnUiThread {
                        toast("আনলক হয়ে সেভ হয়েছে:\n${outFile.absolutePath}")
                    }
                } catch (e: Exception) {
                    runOnUiThread { toast("ডিক্রিপ্ট ত্রুটি: ${e.message}") }
                }
            }.start()
        }
    }

    private fun getFileName(uri: Uri): String {
        contentResolver.query(uri, null, null, null, null)?.use { c ->
            val idx = c.getColumnIndex(OpenableColumns.DISPLAY_NAME)
            if (c.moveToFirst() && idx >= 0) return c.getString(idx)
        }
        return "file_${System.currentTimeMillis()}"
    }

    private fun toast(msg: String) {
        Toast.makeText(this, msg, Toast.LENGTH_LONG).show()
    }
